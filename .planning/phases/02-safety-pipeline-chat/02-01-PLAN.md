Implementation pattern:
```python
class LLMGuardrailScanner:
    def __init__(self):
        self._scanners = {
            "prompt_injection": PromptInjectionScanner(),
            "jailbreak": JailbreakScanner(),
            "toxicity": ToxicityScanner(),
        }

    async def scan(self, content: str, source: str) -> ScannerResult:
        try:
            results = await asyncio.to_thread(self._scan_sync, content)
            findings = self._map_to_findings(results)
            return ScannerResult(findings=findings, has_violations=len(findings) > 0, scanner_name="llm_guardrails")
        except Exception:
            return ScannerResult(findings=[], has_violations=False, scanner_name="llm_guardrails")

    def _scan_sync(self, content: str) -> dict:
        results = {}
        for name, scanner in self._scanners.items():
            try:
                scan_result = scanner.scan(content)
                results[name] = scan_result
            except Exception:
                # Skip this scanner, continue with others
                pass
        return results
```</action>
